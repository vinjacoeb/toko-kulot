import { useCallback, useEffect, useRef, useState } from 'react';

/** Header `scrolled`: true bila halaman digulir melewati ambang (design §3.1). */
export function useScrolled(threshold = 8) {
    const [scrolled, setScrolled] = useState(false);

    useEffect(() => {
        const onScroll = () => setScrolled(window.scrollY > threshold);

        onScroll();
        window.addEventListener('scroll', onScroll, { passive: true });

        return () => window.removeEventListener('scroll', onScroll);
    }, [threshold]);

    return scrolled;
}

/** Menghormati `prefers-reduced-motion` (design §6). */
export function usePrefersReducedMotion() {
    const [reduced, setReduced] = useState(false);

    useEffect(() => {
        const query = window.matchMedia('(prefers-reduced-motion: reduce)');
        const update = () => setReduced(query.matches);

        update();
        query.addEventListener('change', update);

        return () => query.removeEventListener('change', update);
    }, []);

    return reduced;
}

/** Tab browser aktif — carousel berhenti saat tab tidak terlihat. */
export function useDocumentVisible() {
    const [visible, setVisible] = useState(
        typeof document === 'undefined' ? true : document.visibilityState !== 'hidden'
    );

    useEffect(() => {
        const update = () => setVisible(document.visibilityState !== 'hidden');

        document.addEventListener('visibilitychange', update);

        return () => document.removeEventListener('visibilitychange', update);
    }, []);

    return visible;
}

/**
 * design §4.0: loading yang melewati 10 detik wajib berubah jadi error.
 * Mengembalikan true bila ambang waktu terlampaui.
 */
export function useLoadingTimeout(loading, timeoutMs = 10000) {
    const [timedOut, setTimedOut] = useState(false);

    useEffect(() => {
        if (!loading) {
            setTimedOut(false);
            return undefined;
        }

        const id = window.setTimeout(() => setTimedOut(true), timeoutMs);

        return () => window.clearTimeout(id);
    }, [loading, timeoutMs]);

    return timedOut;
}

/** Kunci scroll body saat drawer/modal terbuka (design §4.12). */
export function useBodyScrollLock(active) {
    useEffect(() => {
        if (!active) return undefined;

        const previous = document.body.style.overflow;
        document.body.style.overflow = 'hidden';

        return () => {
            document.body.style.overflow = previous;
        };
    }, [active]);
}

/**
 * Fokus terperangkap di dalam ref selama `active` (drawer/modal).
 * Tab berputar di elemen fokusable pertama-terakhir, Esc diserahkan ke pemanggil.
 */
export function useFocusTrap(ref, active) {
    const previouslyFocused = useRef(null);

    const focusFirst = useCallback(() => {
        const node = ref.current;
        if (!node) return;

        const targets = node.querySelectorAll(
            'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
        );

        if (targets.length > 0) targets[0].focus();
        else node.focus();
    }, [ref]);

    useEffect(() => {
        if (!active) return undefined;

        previouslyFocused.current = document.activeElement;
        focusFirst();

        const onKeyDown = (event) => {
            if (event.key !== 'Tab') return;

            const node = ref.current;
            if (!node) return;

            const targets = Array.from(
                node.querySelectorAll(
                    'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
                )
            ).filter((el) => el.offsetParent !== null);

            if (targets.length === 0) {
                event.preventDefault();
                return;
            }

            const first = targets[0];
            const last = targets[targets.length - 1];

            if (event.shiftKey && document.activeElement === first) {
                event.preventDefault();
                last.focus();
            } else if (!event.shiftKey && document.activeElement === last) {
                event.preventDefault();
                first.focus();
            }
        };

        document.addEventListener('keydown', onKeyDown);

        return () => {
            document.removeEventListener('keydown', onKeyDown);

            if (previouslyFocused.current instanceof HTMLElement) {
                previouslyFocused.current.focus();
            }
        };
    }, [active, focusFirst, ref]);
}

/**
 * Helper format bersama — kontrak: docs/komponen.md §6.
 * Uang selalu integer rupiah dari server (PRD A13); format hanya di klien.
 */

/** Gabung kelas kondisional tanpa dependensi tambahan. */
export const cn = (...classes) => classes.flat().filter(Boolean).join(' ');

/** Integer rupiah -> "Rp149.000". Nilai kosong/tidak valid -> null (pemanggil menampilkan "-"). */
export const rupiah = (value) => {
    if (value === null || value === undefined || value === '') return null;

    const number = Number(value);
    if (!Number.isFinite(number)) return null;

    return `Rp${Math.round(number).toLocaleString('id-ID')}`;
};

/** Angka biasa -> "1.234". Nilai kosong/tidak valid -> "0". */
export const formatNumber = (value) => {
    const number = Number(value);

    return Number.isFinite(number) ? number.toLocaleString('id-ID') : '0';
};

/** Tanggal ISO -> "20 Okt 2026". Nilai kosong/tidak valid -> "-". */
export const formatDate = (value) => {
    if (!value) return '-';

    const date = new Date(value);

    if (Number.isNaN(date.getTime())) return '-';

    return date.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' });
};

/** Tanggal ISO -> "20 Okt 2026, 14.05". Nilai kosong/tidak valid -> "-". */
export const formatDateTime = (value) => {
    if (!value) return '-';

    const date = new Date(value);

    if (Number.isNaN(date.getTime())) return '-';

    return `${date.toLocaleDateString('id-ID', {
        day: '2-digit',
        month: 'short',
        year: 'numeric',
    })}, ${date.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' })}`;
};

/**
 * Kolom JSON dari pengaturan admin bisa datang sebagai array atau string JSON.
 * Selalu kembalikan array agar map() aman.
 */
export const toList = (value) => {
    if (Array.isArray(value)) return value;

    if (typeof value === 'string' && value.trim() !== '') {
        try {
            const parsed = JSON.parse(value);
            return Array.isArray(parsed) ? parsed : [];
        } catch {
            return [];
        }
    }

    return [];
};

/** Objek pengaturan bisa datang sebagai string JSON juga. */
export const toObject = (value) => {
    if (value && typeof value === 'object' && !Array.isArray(value)) return value;

    if (typeof value === 'string' && value.trim() !== '') {
        try {
            const parsed = JSON.parse(value);
            return parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? parsed : {};
        } catch {
            return {};
        }
    }

    return {};
};

/** Nilai server bisa null; semua teks melewati helper ini agar tidak ada "undefined" di layar. */
const asText = (value) => {
    if (value === null || value === undefined) return '';

    return typeof value === 'string' ? value : String(value);
};

/** Angka dari server bisa string; hasil selalu >= 0 dan bukan NaN. */
const asCount = (value) => {
    const number = Number(value ?? 0);

    return Number.isFinite(number) && number > 0 ? number : 0;
};

/** Normalisasi props `filter` dari server; nilai kosong/tak dikenal -> bawaan. */

    status: filter?.status === null || filter?.status === undefined ? 'semua' : String(filter.status),
});

/**
 * Paginator Laravel dikirim datar (`current_page`, `last_page`, `total`, `data`);
 * kontrak §10.1 menuliskannya sebagai `links` + `meta`. Keduanya dibaca supaya
 * halaman tidak bergantung pada satu bentuk saja.
 */
const readPaginator = (paginator) => {
    const source = paginator ?? {};
    const meta = source.meta && typeof source.meta === 'object' ? source.meta : source;

    return {
        rows: Array.isArray(source.data) ? source.data : [],
        currentPage: Math.max(1, Number(meta.current_page) || 1),
        lastPage: Math.max(1, Number(meta.last_page) || 1),
        total: asCount(meta.total),
    };
};

import React, { useEffect, useRef } from 'react';
import { cn } from '../lib/format';
import { useBodyScrollLock, useFocusTrap } from '../lib/hooks';
import { CloseIcon } from './icons';

const SIZES = {
    md: 'max-w-modal',
    lg: 'max-w-modalLg',
};

/**
 * Modal dasar — docs/design.md §4.24 (turunan: `ConfirmDialog`).
 * Overlay gelap, Esc menutup, klik overlay menutup, fokus terperangkap,
 * scroll body dikunci, `aria-modal="true"`.
 */
export default function Modal({
    open = false,
    title = null,
    description = null,
    children = null,
    footer = null,
    onClose,
    size = 'md',
    closeLabel = 'Tutup',
}) {
    const panelRef = useRef(null);

    useBodyScrollLock(open);
    useFocusTrap(panelRef, open);

    useEffect(() => {
        if (!open) return undefined;

        const onKeyDown = (event) => {
            if (event.key === 'Escape') onClose?.();
        };

        document.addEventListener('keydown', onKeyDown);

        return () => document.removeEventListener('keydown', onKeyDown);
    }, [open, onClose]);

    if (!open) return null;

    return (
        <div
            className="fixed inset-0 z-modal flex items-center justify-center bg-black/50 p-4"
            onMouseDown={(event) => {
                if (event.target === event.currentTarget) onClose?.();
            }}
        >
            <div
                ref={panelRef}
                role="dialog"
                aria-modal="true"
                aria-labelledby={title ? 'modal-title' : undefined}
                tabIndex={-1}
                className={cn(
                    'w-full rounded-xl bg-white p-6 shadow-raised animate-modal-in',
                    SIZES[size] ?? SIZES.md
                )}
            >
                <div className="flex items-start justify-between gap-4">
                    <div className="min-w-0">
                        {title && (
                            <h2 id="modal-title" className="text-h3 text-ink">
                                {title}
                            </h2>
                        )}
                        {description && <p className="mt-1 text-meta text-ink-soft">{description}</p>}
                    </div>
                    {onClose && (
                        <button
                            type="button"
                            onClick={onClose}
                            aria-label={closeLabel}
                            className="-mr-2 -mt-1 inline-flex h-touch w-touch items-center justify-center rounded-md text-ink-soft transition-colors duration-hover hover:bg-surface"
                        >
                            <CloseIcon size={20} />
                        </button>
                    )}
                </div>

                {children && <div className="mt-4 text-body text-ink">{children}</div>}

/**
 * Tabel data admin — docs/design.md §4.28, docs/komponen.md §4.
 * Header sticky `h-tableHead bg-surface`, baris `h-tableRow`, teks `text-meta`.
 * Memuat: 8 baris skeleton. Kosong dan galat adalah dua keadaan berbeda.
 */
export default function DataTable({
    columns = [],
    rows = [],
    rowKey = (row, index) => row?.id ?? index,
    loading = false,
    error = null,
    emptyTitle = 'Belum ada data',
    emptyDescription = null,
    emptyActionLabel = null,
    emptyActionHref = null,
    onRetry = null,
    onRowClick = null,
    sort = null,
    onSort = null,
    caption = null,
    className = '',
}) {
    const list = Array.isArray(rows) ? rows : [];
    const hasRows = list.length > 0;

    const cellAlign = (column) =>
        column.align === 'right' ? 'text-right' : column.align === 'center' ? 'text-center' : 'text-left';

    const sortState = (column) => {
        if (!sort || sort.key !== column.key) return 'none';
        return sort.dir === 'desc' ? 'descending' : 'ascending';
    };

    return (
        <div className={cn('overflow-hidden rounded-lg border border-line bg-white', className)}>
            {error ? (
                <ErrorState
                    title="Gagal memuat data"
                    description={typeof error === 'string' ? error : 'Periksa koneksi Anda lalu coba lagi.'}
                    onRetry={onRetry}
                    supportHref={null}
                />
            ) : loading ? (
                <div className="flex flex-col" aria-busy="true">
                    <div className="h-tableHead w-full border-b border-line bg-surface" aria-hidden="true" />
                    {Array.from({ length: 8 }).map((_, index) => (
                        <div key={index} className="flex items-center gap-4 border-b border-line px-4 last:border-b-0">
                            <Skeleton variant="row" />
                        </div>
                    ))}
                </div>
            ) : !hasRows ? (
                <EmptyState
                    title={emptyTitle}
                    description={emptyDescription}
                    actionLabel={emptyActionLabel}
                    actionHref={emptyActionHref}
                />
            ) : (
                <div className="overflow-x-auto">
                    <table className="w-full border-collapse text-meta">
                        {caption && <caption className="sr-only">{caption}</caption>}
                        <thead>
                            <tr className="bg-surface">
                                {columns.map((column) => {
                                    const isSorted = sort && sort.key === column.key;
                                    const sortable = Boolean(onSort && column.sortable !== false);

                                        <th

                                            scope="col"

                                                >

    const classes = cn(
        VARIANTS[variant] ?? VARIANTS.primary,
        size === 'sm' && variant !== 'link' ? SIZES.sm : null,
        variant === 'link' && size === 'sm' ? 'text-meta' : null,
        fullWidth ? 'w-full' : null,
        'whitespace-nowrap',
        className
    );

    const content = loading ? (
        <>
            <SpinnerIcon size={16} />
            <span>Memproses…</span>
        </>
    ) : (
        <>
            {icon}
            <span>{children}</span>
        </>
    );

    if (as === 'a') {
        return (
            <a href={href} className={classes} {...rest}>
                {content}
            </a>
        );
    }

    if (as === Link || as === 'Link' || href) {
        return (
            <Link href={href} className={classes} aria-busy={loading || undefined} {...rest}>
                {content}
            </Link>
        );
    }

    return (
        <button
            type={type}
            className={classes}
            disabled={disabled || loading}
            aria-busy={loading || undefined}
            {...rest}
        >
            {content}
        </button>
    );
}

/** SearchInput - docs/design.md 4.19. Debounce 300ms sebelum memanggil onChange. */
export default function SearchInput({
    value = '',
    onChange = null,
    placeholder = 'Cari produk',
    label = 'Cari produk',
    loading = false,
    className = '',
}) {
    const [text, setText] = useState(value);
    const mounted = useRef(false);

    useEffect(() => {
        setText(value);
    }, [value]);

    useEffect(() => {
        if (!mounted.current) {

            return undefined;
        }
        const timer = window.setTimeout(() => {

        }, 300);
        return () => window.clearTimeout(timer);
    }, [text]);

    return (
        <div className={cn('relative', className)}>
            <span className="pointer-events-none absolute left-3.5 top-1/2 -translate-y-1/2 text-slate-400" aria-hidden="true">
                <SearchIcon width={18} height={18} />
            </span>
            <input
                type="search"
                value={text}
                onChange={(event) => setText(event.target.value)}
                placeholder={placeholder}
                aria-label={label}
                aria-busy={loading ? 'true' : undefined}
                className="h-field w-full rounded-md border border-line bg-white pl-10 pr-10 text-body text-ink transition-colors duration-hover placeholder:text-slate-400 focus:border-brand focus:outline-none focus:ring-[3px] focus:ring-brand/15"
            />

                <button
                    type="button"
                    onClick={() => setText('')}

--- Pagination ---
import React from 'react';
import { cn } from '../lib/format';

function buildPages(current, last) {
    const pages = [];
    const push = (value) => {
        if (value >= 1 && value <= last && !pages.includes(value)) pages.push(value);
    };
    push(1);
    for (let page = current - 1; page <= current + 1; page += 1) push(page);
    push(last);
    return pages.sort((a, b) => a - b);
}

function Chevron({ direction }) {
    return (
        <svg
            viewBox="0 0 24 24"
            width="16"
            height="16"
            fill="none"
            stroke="currentColor"
            strokeWidth="1.5"
            strokeLinecap="round"
            strokeLinejoin="round"
            aria-hidden="true"
        >
            {direction === 'left' ? <path d="M15 6l-6 6 6 6" /> : <path d="M9 6l6 6-6 6" />}
        </svg>
    );
}

/** Pagination - docs/design.md 4.22. Tombol 36x36, aktif bg-brand-700. */
export default function Pagination({ currentPage = 1, lastPage = 1, onPageChange = null, className = '' }) {
    const current = Math.max(1, Number(currentPage) || 1);
    const last = Math.max(1, Number(lastPage) || 1);
    if (last <= 1 || typeof onPageChange !== 'function') return null;

    const go = (page) => {
        if (page < 1 || page > last || page === current) return;
        onPageChange(page);
    };

    const base =
        'inline-flex h-9 w-9 items-center justify-center rounded-sm text-meta transition-colors duration-hover focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand focus-visible:ring-offset-2';
    const pages = buildPages(current, last);

    return (
        <nav aria-label="Navigasi halaman" className={cn('flex items-center justify-center gap-1', className)}>
            <button
                type="button"
                onClick={() => go(current - 1)}
                disabled={current <= 1}
                aria-label="Halaman sebelumnya"
                className={cn(base, 'text-slate-400 hover:bg-surface disabled:cursor-not-allowed disabled:opacity-45')}
            >
                <Chevron direction="left" />
            </button>

            <span className="px-2 text-meta text-ink-soft sm:hidden" aria-live="polite">
                {current} / {last}
            </span>

            <div className="hidden items-center gap-1 sm:flex">
                {pages.map((page, index) => {
                    const previous = pages[index - 1];
                    const gap = previous && page - previous > 1;
                    return (
                        <React.Fragment key={page}>
                            {gap && (
                                <span className="px-1 text-meta text-slate-400" aria-hidden="true">
                                    ...
                                </span>
                            )}
                            <button
                                type="button"
                                onClick={() => go(page)}
                                aria-current={page === current ? 'page' : undefined}
                                aria-label={'Halaman ' + page}
                                className={cn(
                                    base,
                                    page === current ? 'bg-brand-700 text-white' : 'text-ink hover:bg-surface'
                                )}
                            >
                                {page}
                            </button>
                        </React.Fragment>
                    );
                })}
            </div>

            <button
                type="button"
                onClick={() => go(current + 1)}
                disabled={current >= last}
                aria-label="Halaman berikutnya"
                className={cn(base, 'text-slate-400 hover:bg-surface disabled:cursor-not-allowed disabled:opacity-45')}
            >
                <Chevron direction="right" />
            </button>
        </nav>
    );
}

import React, { useEffect } from 'react';
import { cn } from '../lib/format';
import { AlertIcon, CheckIcon, CloseIcon, SearchIcon } from './icons';

const TONES = {
    info: { bar: 'border-l-brand', icon: SearchIcon, color: 'text-brand' },
    success: { bar: 'border-l-success', icon: CheckIcon, color: 'text-success' },
    warning: { bar: 'border-l-warning', icon: AlertIcon, color: 'text-warning' },
    error: { bar: 'border-l-danger', icon: AlertIcon, color: 'text-danger' },
};

/**
 * Notifikasi mengambang — docs/komponen.md §3, design §6 (`animate-fade-up`, `z-toast`).
 * Penumpukan maksimal 3 diatur pemanggil.
 */
export default function Toast({
    tone = 'info',
    title = null,
    message = null,
    duration = 4000,
    onClose = null,
}) {
    const config = TONES[tone] ?? TONES.info;
    const Icon = config.icon;

    useEffect(() => {
        if (!duration || !onClose) return undefined;

        const id = window.setTimeout(onClose, duration);

        return () => window.clearTimeout(id);
    }, [duration, onClose]);

    return (
        <div
            role={tone === 'error' ? 'alert' : 'status'}
            className={cn(
                'pointer-events-auto flex w-full max-w-[360px] items-start gap-3 rounded-md border border-line border-l-[3px] bg-white p-4 shadow-raised animate-fade-up',
                config.bar
            )}
        >
            <span className={cn('mt-0.5 shrink-0', config.color)} aria-hidden="true">
                <Icon size={20} />
            </span>
            <div className="min-w-0 flex-1">
                {title && <p className="text-label text-ink">{title}</p>}
                {message && <p className="text-meta text-ink-soft">{message}</p>}
            </div>
            {onClose && (
                <button
                    type="button"
                    onClick={onClose}
                    aria-label="Tutup notifikasi"
                    className="shrink-0 rounded-sm p-1 text-ink-soft transition-colors duration-hover hover:bg-surface"
                >
                    <CloseIcon size={16} />
                </button>
            )}
        </div>
    );
}
