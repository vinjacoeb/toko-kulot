## 4. Inventaris Komponen

Setiap komponen: struktur, ukuran, state. Wajib jadi komponen React reusable di `resources/js/Components/`. API props & checklist implementasi ada di `docs/komponen.md`.

### 4.0 Matriks state universal

Berlaku untuk **setiap** komponen yang mengambil data atau menerima aksi pengguna. QA menolak komponen yang tidak punya state di bawah.

| State | Pemicu | Tampilan wajib | Kelas/token acuan |
|---|---|---|---|
| `default` | data ada | konten normal | — |
| `loading` | request berjalan | skeleton mengikuti dimensi asli (bukan spinner penuh halaman), `aria-busy="true"` | `animate-pulse-slow` |
| `empty` | data kosong & bukan error | ilustrasi garis + teks penjelas + CTA keluar dari kondisi kosong | `text-slate-400`, §4.23 `EmptyState` |
| `error` | request gagal | pesan di dalam layout (bukan halaman putih) + tombol "Coba lagi" | `border-danger`, `text-danger`, §4.23 `ErrorState` |
| `disabled` | aksi tidak tersedia (stok 0, belum login) | opacity .45 + `cursor-not-allowed` + alasan terlihat | `disabled:opacity-45` |
| `focus-visible` | navigasi keyboard | ring 2px `brand` offset 2px | `ring-2 ring-brand ring-offset-2` |
| `hover` | pointer di atas elemen | sesuai komponen | `duration-hover` |

Aturan tambahan: `loading` yang melewati **10 detik** wajib berubah jadi `error` + tombol "Coba lagi" (mencegah layar menggantung); skeleton tidak boleh dipakai untuk menutupi error.

function Thumb({ src, size = 48, className = '' }) {
    const [failed, setFailed] = useState(false);

    useEffect(() => {
        setFailed(false);
    }, [src]);

    const style = { width: `${size}px`, height: `${size}px` };

    if (!src || failed) {
        return (
            <span
                style={style}
                className={cn('flex shrink-0 items-center justify-center rounded-md border border-line bg-surface text-slate-400', className)}
            >
                <ImageOffIcon size={18} />
            </span>
        );
    }

    return (
        <img
            src={src}
            alt=""
            style={style}
            loading="lazy"
            onError={() => setFailed(true)}
            className={cn('shrink-0 rounded-md border border-line object-cover', className)}
        />
    );
}

function ImageField({
    id,
    label,
    hint = null,
    required = false,
    accept,
    maxKb,
    extensions,
    file = null,
    currentPath = null,
    error = null,
    message,
    onSelect,
}) {
    const [objectUrl, setObjectUrl] = useState(null);

    useEffect(() => {
        if (!file) {
            setObjectUrl(null);
            return undefined;
        }
        const url = URL.createObjectURL(file);
        setObjectUrl(url);
        return () => URL.revokeObjectURL(url);
    }, [file]);

    const describedBy = error ? `${id}-error` : hint ? `${id}-hint` : undefined;
    const sizeLabel = maxKb >= 1024 ? `${Math.round(maxKb / 1024)} MB` : `${maxKb} KB`;

    return (
        <Field label={label} htmlFor={id} required={required} hint={hint} error={error}>
            <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:gap-4">
                <Thumb src={objectUrl || currentPath || null} size={112} className="rounded-lg" />
                <div className="flex min-w-0 flex-1 flex-col gap-2">
                    <input
                        id={id}
                        type="file"
                        accept={accept}
                        aria-describedby={describedBy}
                        aria-invalid={error ? 'true' : undefined}
                        onChange={(event) => {
                            const picked = event.target.files && event.target.files[0] ? event.target.files[0] : null;
                            onSelect(picked);
                            event.target.value = '';
                        }}
                        className={cn(
                            'field h-auto cursor-pointer py-2 text-meta file:mr-3 file:rounded-sm file:border-0 file:bg-surface file:px-3 file:py-1.5 file:text-meta file:font-semibold file:text-ink hover:file:bg-slate-200',
                            error && 'field-error'
                        )}
                    />
                    <p className="text-meta text-slate-400">
                        {message} Maksimal {sizeLabel}.
                    </p>
                    {file ? (
                        <div className="flex flex-wrap items-center gap-2">
                            <span className="truncate text-meta text-ink-soft">{file.name}</span>
                            <Button variant="ghost" size="sm" onClick={() => onSelect(null)}>
                                Batalkan pilihan
                            </Button>
                        </div>
                    ) : null}
                </div>
            </div>
        </Field>
    );
}

function ConfirmDialog({ open, title, description, confirmLabel, processing, onConfirm, onCancel }) {
    return (
        <Modal
            open={open}
            title={title}
            description={description}
            onClose={processing ? () => {} : onCancel}
            size="md"
            footer={
                <div className="flex flex-col gap-2 sm:flex-row sm:justify-end">
                    <Button variant="secondary" onClick={onCancel} disabled={processing}>
                        Batal
                    </Button>
                    <Button variant="danger" onClick={onConfirm} loading={processing}>
                        {confirmLabel}
                    </Button>
                </div>
            }
        >
            <p className="text-body text-ink-soft">Tindakan ini tidak bisa dibatalkan.</p>
        </Modal>
    );
}

import React, { useEffect, useRef } from 'react';
import { cn } from '../lib/format';
import { useBodyScrollLock, useFocusTrap } from '../lib/hooks';
import { CloseIcon } from './icons';

const SIZES = {
    md: 'max-w-modal',
    lg: 'max-w-modalLg',
};

/**
 * Modal dasar — docs/design.md §4.24 (turunan: `ConfirmDialog`).
 * Overlay gelap, Esc menutup, klik overlay menutup, fokus terperangkap,
 * scroll body dikunci, `aria-modal="true"`.
 */
export default function Modal({
    open = false,
    title = null,
    description = null,
    children = null,
    footer = null,
    onClose,
    size = 'md',
    closeLabel = 'Tutup',
}) {
    const panelRef = useRef(null);

    useBodyScrollLock(open);
    useFocusTrap(panelRef, open);

    useEffect(() => {
        if (!open) return undefined;

        const onKeyDown = (event) => {
            if (event.key === 'Escape') onClose?.();
        };

        document.addEventListener('keydown', onKeyDown);

        return () => document.removeEventListener('keydown', onKeyDown);
    }, [open, onClose]);

    if (!open) return null;

    return (
        <div
            className="fixed inset-0 z-modal flex items-center justify-center bg-black/50 p-4"
            onMouseDown={(event) => {
                if (event.target === event.currentTarget) onClose?.();
            }}
        >
            <div
                ref={panelRef}
                role="dialog"
                aria-modal="true"
                aria-labelledby={title ? 'modal-title' : undefined}
                tabIndex={-1}
                className={cn(
                    'w-full rounded-xl bg-white p-6 shadow-raised animate-modal-in',
                    SIZES[size] ?? SIZES.md
                )}
            >
                <div className="flex items-start justify-between gap-4">
                    <div className="min-w-0">
                        {title && (
                            <h2 id="modal-title" className="text-h3 text-ink">
                                {title}
                            </h2>
                        )}
                        {description && <p className="mt-1 text-meta text-ink-soft">{description}</p>}
                    </div>
                    {onClose && (
                        <button
                            type="button"
                            onClick={onClose}
                            aria-label={closeLabel}
                            className="-mr-2 -mt-1 inline-flex h-touch w-touch items-center justify-center rounded-md text-ink-soft transition-colors duration-hover hover:bg-surface"
                        >
                            <CloseIcon size={20} />
                        </button>
                    )}
                </div>

                {children && <div className="mt-4 text-body text-ink">{children}</div>}

import React from 'react';
import { cn } from '../lib/format';

function buildPages(current, last) {
    const pages = [];
    const push = (value) => {
        if (value >= 1 && value <= last && !pages.includes(value)) pages.push(value);
    };
    push(1);
    for (let page = current - 1; page <= current + 1; page += 1) push(page);
    push(last);
    return pages.sort((a, b) => a - b);
}

function Chevron({ direction }) {
    return (
        <svg
            viewBox="0 0 24 24"
            width="16"
            height="16"
            fill="none"
            stroke="currentColor"
            strokeWidth="1.5"
            strokeLinecap="round"
            strokeLinejoin="round"
            aria-hidden="true"
        >
            {direction === 'left' ? <path d="M15 6l-6 6 6 6" /> : <path d="M9 6l6 6-6 6" />}
        </svg>
    );
}

/** Pagination - docs/design.md 4.22. Tombol 36x36, aktif bg-brand-700. */
export default function Pagination({ currentPage = 1, lastPage = 1, onPageChange = null, className = '' }) {
    const current = Math.max(1, Number(currentPage) || 1);
    const last = Math.max(1, Number(lastPage) || 1);
    if (last <= 1 || typeof onPageChange !== 'function') return null;

    const go = (page) => {
        if (page < 1 || page > last || page === current) return;
        onPageChange(page);
    };

    const base =
        'inline-flex h-9 w-9 items-center justify-center rounded-sm text-meta transition-colors duration-hover focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand focus-visible:ring-offset-2';
    const pages = buildPages(current, last);

    return (
        <nav aria-label="Navigasi halaman" className={cn('flex items-center justify-center gap-1', className)}>
            <button
                type="button"
                onClick={() => go(current - 1)}
                disabled={current <= 1}
                aria-label="Halaman sebelumnya"
                className={cn(base, 'text-slate-400 hover:bg-surface disabled:cursor-not-allowed disabled:opacity-45')}
            >
                <Chevron direction="left" />
            </button>

            <span className="px-2 text-meta text-ink-soft sm:hidden" aria-live="polite">
                {current} / {last}
            </span>

            <div className="hidden items-center gap-1 sm:flex">
                {pages.map((page, index) => {
                    const previous = pages[index - 1];
                    const gap = previous && page - previous > 1;
                    return (
                        <React.Fragment key={page}>
                            {gap && (
                                <span className="px-1 text-meta text-slate-400" aria-hidden="true">
                                    ...
                                </span>
                            )}
                            <button
                                type="button"
                                onClick={() => go(page)}
                                aria-current={page === current ? 'page' : undefined}
                                aria-label={'Halaman ' + page}
                                className={cn(
                                    base,
                                    page === current ? 'bg-brand-700 text-white' : 'text-ink hover:bg-surface'
                                )}
                            >
                                {page}
                            </button>
                        </React.Fragment>

import React from 'react';

/**
 * Breadcrumb - docs/design.md 4.13.
 * items = [{ label, href }]; item terakhir aria-current="page".
 * Mobile: maksimal 2 level, level tengah diganti elipsis.
 */
export default function Breadcrumb({ items = [], className = '' }) {
    const list = (Array.isArray(items) ? items : []).filter((item) => item && item.label);
    if (list.length === 0) return null;

    const first = list[0];
    const last = list[list.length - 1];
    const middle = list.slice(1, -1);

    const renderItem = (item, isLast) =>
        isLast || !item.href ? (
            <span className="text-ink" aria-current="page">
                {item.label}
            </span>
        ) : (
            <Link href={item.href} className="transition-colors duration-hover hover:text-ink">
                {item.label}
            </Link>
        );

    return (
        <nav aria-label="Breadcrumb" className={cn('h-12', className)}>
            <ol className="flex h-full items-center gap-2 text-meta text-slate-400">
                <li className="flex items-center gap-2">{renderItem(first, list.length === 1)}</li>
                {list.length > 1 && (
                    <li className="flex items-center gap-2" aria-hidden="true">
                        <span>/</span>
                    </li>
                )}
                {middle.length > 0 && (
                    <li className="flex items-center gap-2 sm:hidden" aria-hidden="true">
                        <span>...</span>
                    </li>
                )}
                {list.length > 1 && (
                    <li className="flex items-center gap-2">{renderItem(last, true)}</li>
                )}
            </ol>
        </nav>
    );
}

/**
 * Helper format bersama — kontrak: docs/komponen.md §6.
 * Uang selalu integer rupiah dari server (PRD A13); format hanya di klien.
 */

/** Gabung kelas kondisional tanpa dependensi tambahan. */
export const cn = (...classes) => classes.flat().filter(Boolean).join(' ');

/** Integer rupiah -> "Rp149.000". Nilai kosong/tidak valid -> null (pemanggil menampilkan "-"). */
export const rupiah = (value) => {
    if (value === null || value === undefined || value === '') return null;

    const number = Number(value);
    if (!Number.isFinite(number)) return null;

    return `Rp${Math.round(number).toLocaleString('id-ID')}`;
};

/** Angka biasa -> "1.234". Nilai kosong/tidak valid -> "0". */
export const formatNumber = (value) => {
    const number = Number(value);

    return Number.isFinite(number) ? number.toLocaleString('id-ID') : '0';
};

/** Tanggal ISO -> "20 Okt 2026". Nilai kosong/tidak valid -> "-". */
export const formatDate = (value) => {
    if (!value) return '-';

    const date = new Date(value);

    if (Number.isNaN(date.getTime())) return '-';

    return date.toLocaleDateString('id-ID', { day: '2-digit', month: 'short', year: 'numeric' });
};

/** Tanggal ISO -> "20 Okt 2026, 14.05". Nilai kosong/tidak valid -> "-". */
export const formatDateTime = (value) => {
    if (!value) return '-';

    const date = new Date(value);

    if (Number.isNaN(date.getTime())) return '-';

    return `${date.toLocaleDateString('id-ID', {
        day: '2-digit',
        month: 'short',
        year: 'numeric',
    })}, ${date.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' })}`;
};

/**
 * Kolom JSON dari pengaturan admin bisa datang sebagai array atau string JSON.
 * Selalu kembalikan array agar map() aman.
 */
export const toList = (value) => {
    if (Array.isArray(value)) return value;

    if (typeof value === 'string' && value.trim() !== '') {
        try {
            const parsed = JSON.parse(value);
            return Array.isArray(parsed) ? parsed : [];
        } catch {
            return [];
        }
    }

    return [];
};

/** Objek pengaturan bisa datang sebagai string JSON juga. */
export const toObject = (value) => {
    if (value && typeof value === 'object' && !Array.isArray(value)) return value;

    if (typeof value === 'string' && value.trim() !== '') {
        try {
            const parsed = JSON.parse(value);
            return parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? parsed : {};
        } catch {
            return {};
        }
    }

    return {};
};

import { useCallback, useEffect, useRef, useState } from 'react';

/** Header `scrolled`: true bila halaman digulir melewati ambang (design §3.1). */
export function useScrolled(threshold = 8) {
    const [scrolled, setScrolled] = useState(false);

    useEffect(() => {
        const onScroll = () => setScrolled(window.scrollY > threshold);

        onScroll();
        window.addEventListener('scroll', onScroll, { passive: true });

        return () => window.removeEventListener('scroll', onScroll);
    }, [threshold]);

    return scrolled;
}

/** Menghormati `prefers-reduced-motion` (design §6). */
export function usePrefersReducedMotion() {
    const [reduced, setReduced] = useState(false);

    useEffect(() => {
        const query = window.matchMedia('(prefers-reduced-motion: reduce)');
        const update = () => setReduced(query.matches);

        update();
        query.addEventListener('change', update);

        return () => query.removeEventListener('change', update);
    }, []);

    return reduced;
}

/** Tab browser aktif — carousel berhenti saat tab tidak terlihat. */
export function useDocumentVisible() {
    const [visible, setVisible] = useState(
        typeof document === 'undefined' ? true : document.visibilityState !== 'hidden'
    );

    useEffect(() => {
        const update = () => setVisible(document.visibilityState !== 'hidden');

        document.addEventListener('visibilitychange', update);

        return () => document.removeEventListener('visibilitychange', update);
    }, []);

    return visible;
}

/**

 * Mengembalikan true bila ambang waktu terlampaui.

export function useLoadingTimeout(loading, timeoutMs = 10000) {

        if (!loading) {
            setTimedOut(false);
            return undefined;

/**
 * Fokus terperangkap di dalam ref selama `active` (drawer/modal).
 * Tab berputar di elemen fokusable pertama-terakhir, Esc diserahkan ke pemanggil.
 */
export function useFocusTrap(ref, active) {
    const previouslyFocused = useRef(null);

    const focusFirst = useCallback(() => {
        const node = ref.current;
        if (!node) return;

        const targets = node.querySelectorAll(
            'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
        );

        if (targets.length > 0) targets[0].focus();
        else node.focus();
    }, [ref]);

    useEffect(() => {
        if (!active) return undefined;

        previouslyFocused.current = document.activeElement;
        focusFirst();

        const onKeyDown = (event) => {
            if (event.key !== 'Tab') return;

            const node = ref.current;
            if (!node) return;

            const targets = Array.from(
                node.querySelectorAll(
                    'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
                )
            ).filter((el) => el.offsetParent !== null);

            if (targets.length === 0) {
                event.preventDefault();
                return;
            }

            const first = targets[0];
            const last = targets[targets.length - 1];

            if (event.shiftKey && document.activeElement === first) {
                event.preventDefault();
                last.focus();
            } else if (!event.shiftKey && document.activeElement === last) {
                event.preventDefault();
                first.focus();
            }
        };

        document.addEventListener('keydown', onKeyDown);

        return () => {
            document.removeEventListener('keydown', onKeyDown);

            if (previouslyFocused.current instanceof HTMLElement) {
                previouslyFocused.current.focus();
            }
        };
    }, [active, focusFirst, ref]);
}
