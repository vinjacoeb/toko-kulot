<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Admin\Concerns\ListProps;
use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\StoreCategoryRequest;
use App\Http\Requests\Admin\UpdateCategoryRequest;
use App\Models\Category;
use App\Support\MediaStorage;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;
use Inertia\Inertia;
use Inertia\Response;

/**
 * Kategori admin (PRD §5.2, kontrak props §12.8–§12.11).
 *
 * Tidak ada halaman tambah/ubah tersendiri: form kategori dikerjakan inline pada
 * halaman daftar, jadi controller ini hanya punya index + tulisan redirect.
 */
class CategoryController extends Controller
{
    use ListProps;

    public function index(Request $request): Response
    {
        $filter = [
            'q' => $this->teks($request->query('q')),
            'status' => $this->pilihan($request->query('status'), ['aktif', 'nonaktif'], 'semua'),
        ];

        $query = Category::query()->withCount('products');

        if ($filter['q'] !== null) {
            $like = '%'.$filter['q'].'%';
            $query->where(fn ($where) => $where
                ->where('name', 'like', $like)
                ->orWhere('slug', 'like', $like));
        }

        if ($filter['status'] === 'aktif') {
            $query->where('is_active', true);
        } elseif ($filter['status'] === 'nonaktif') {
            $query->where('is_active', false);
        }

        $kategori = $query
            ->orderBy('sort_order')
            ->orderBy('name')
            ->paginate(15)
            ->withQueryString();

        return Inertia::render('Admin/Categories/Index', [
            'kategori' => [
                'data' => collect($kategori->items())->map(fn (Category $row) => $this->row($row))->all(),
                'links' => $kategori->linkCollection()->all(),
                'meta' => $this->meta($kategori),
            ],
            'filter' => $filter,
            'ringkasan' => [
                'total' => Category::query()->count(),
                'aktif' => Category::query()->where('is_active', true)->count(),
                'nonaktif' => Category::query()->where('is_active', false)->count(),
            ],
        ]);
    }

    public function store(StoreCategoryRequest $request): RedirectResponse
    {
        $data = $request->validated();
        $image = $request->file('image');

        Category::query()->create([
            'name' => trim((string) $data['name']),
            'slug' => $this->uniqueSlug((string) ($data['slug'] ?? ''), (string) $data['name'], null),
            'description' => $this->teks($data['description'] ?? null),
            'image' => $image === null ? null : MediaStorage::store($image, 'kategori'),
            'sort_order' => (int) ($data['sort_order'] ?? 0),
            'is_active' => $this->flag($data['is_active'] ?? null, true),
        ]);

        return redirect()
            ->route('admin.categories.index')
            ->with('success', 'Kategori disimpan.');
    }

    public function update(UpdateCategoryRequest $request, Category $category): RedirectResponse
    {
        $data = $request->validated();
        $image = $request->file('image');
        $imageLama = $category->image;

        $category->forceFill([
            'name' => trim((string) $data['name']),
            'slug' => $this->uniqueSlug((string) ($data['slug'] ?? ''), (string) $data['name'], $category),
            'description' => $this->teks($data['description'] ?? null),
            'image' => $image === null ? $imageLama : MediaStorage::store($image, 'kategori'),
            'sort_order' => (int) ($data['sort_order'] ?? $category->sort_order),
            'is_active' => $this->flag($data['is_active'] ?? null, (bool) $category->is_active),
        ])->save();

        if ($image !== null && $imageLama !== null) {
            MediaStorage::delete($imageLama);
        }

        return redirect()
            ->route('admin.categories.index')
            ->with('success', 'Kategori diperbarui.');
    }

    public function destroy(Category $category): RedirectResponse
    {
        $jumlahProduk = $category->products()->count();

        if ($jumlahProduk > 0) {
            throw ValidationException::withMessages([
                'kategori' => sprintf(
                    'Kategori masih dipakai %d produk — pindahkan atau hapus produknya dulu.',
                    $jumlahProduk,
                ),
            ]);
        }

        MediaStorage::delete($category->image);
        $category->delete();

        return redirect()
            ->route('admin.categories.index')
            ->with('success', 'Kategori dihapus.');
    }

    /**
     * @return array<string, mixed>
     */
    private function row(Category $category): array
    {
        return [
            'id' => $category->id,
            'name' => $category->name,
            'slug' => $category->slug,
            'description' => $category->description,
            'image' => $category->image,
            'sort_order' => (int) $category->sort_order,
            'is_active' => (bool) $category->is_active,
            'product_count' => (int) ($category->products_count ?? 0),
            'created_at' => $category->created_at?->toIso8601String(),
            'updated_at' => $category->updated_at?->toIso8601String(),
        ];
    }

    /**
     * Slug dari input atau dari nama; ditambah akhiran angka bila sudah dipakai.
     */
    private function uniqueSlug(string $slug, string $name, ?Category $category): string
    {
        $base = Str::slug($slug !== '' ? $slug : $name);

        if ($base === '') {
            throw ValidationException::withMessages([
                'slug' => 'Slug tidak bisa dibuat dari nama kategori — isi slug manual.',
            ]);
        }

        $candidate = $base;
        $suffix = 2;

        while (Category::query()
            ->where('slug', $candidate)
            ->when($category, fn ($query) => $query->whereKeyNot($category->id))
            ->exists()) {
            $candidate = $base.'-'.$suffix;
            $suffix++;
        }

        return $candidate;
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Admin\Concerns\ListProps;
use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\StoreAccountRequest;
use App\Http\Requests\Admin\UpdateAccountRequest;
use App\Models\Admin;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\ValidationException;
use Inertia\Inertia;
use Inertia\Response;

/**
 * Akun admin (PRD §5.2, kontrak props §12.19–§12.22).
 *
 * Penjaga terakhir: tidak boleh menghapus/menonaktifkan akun sendiri, dan harus
 * selalu tersisa minimal satu owner aktif.
 */
class AccountController extends Controller
{
    use ListProps;

    /** @var list<string> */
    private const ROLES = ['owner', 'admin', 'staff'];

    public function index(Request $request): Response
    {
        $filter = [
            'q' => $this->teks($request->query('q')),
            'role' => $this->pilihan($request->query('role'), self::ROLES, 'semua'),
            'status' => $this->pilihan($request->query('status'), ['aktif', 'nonaktif'], 'semua'),
        ];

        $query = Admin::query();

        if ($filter['q'] !== null) {
            $like = '%'.$filter['q'].'%';
            $query->where(fn ($where) => $where
                ->where('name', 'like', $like)
                ->orWhere('email', 'like', $like));
        }

        if ($filter['role'] !== 'semua') {
            $query->where('role', $filter['role']);
        }

        if ($filter['status'] === 'aktif') {
            $query->where('is_active', true);
        } elseif ($filter['status'] === 'nonaktif') {
            $query->where('is_active', false);
        }

        $akun = $query
            ->orderByRaw("case role when 'owner' then 0 when 'admin' then 1 else 2 end")
            ->orderBy('name')
            ->paginate(15)
            ->withQueryString();

        $me = $request->user('admin');

        return Inertia::render('Admin/Accounts/Index', [
            'akun' => [
                'data' => collect($akun->items())
                    ->map(fn (Admin $row) => $this->row($row, $me?->id))
                    ->all(),
                'links' => $akun->linkCollection()->all(),
                'meta' => $this->meta($akun),
            ],
            'filter' => $filter,
            'ringkasan' => [
                'total' => Admin::query()->count(),
                'owner' => Admin::query()->where('role', 'owner')->count(),
                'admin' => Admin::query()->where('role', 'admin')->count(),
                'staff' => Admin::query()->where('role', 'staff')->count(),
                'aktif' => Admin::query()->where('is_active', true)->count(),
            ],
        ]);
    }

    public function store(StoreAccountRequest $request): RedirectResponse
    {
        $data = $request->validated();

        Admin::query()->create([
            'name' => trim((string) $data['name']),
            'email' => strtolower(trim((string) $data['email'])),
            'password' => (string) $data['password'],
            'role' => (string) $data['role'],
            'is_active' => $this->flag($data['is_active'] ?? null, true),
        ]);

        return redirect()
            ->route('admin.accounts.index')
            ->with('success', 'Akun admin disimpan.');
    }

    public function update(UpdateAccountRequest $request, Admin $admin): RedirectResponse
    {
        $data = $request->validated();
        $me = $request->user('admin');
        $role = (string) $data['role'];
        $aktif = $this->flag($data['is_active'] ?? null, (bool) $admin->is_active);

        if ($me !== null && $me->id === $admin->id && $role !== $admin->role) {
            throw ValidationException::withMessages([
                'role' => 'Tidak bisa mengubah role akun sendiri.',
            ]);
        }

        if ($me !== null && $me->id === $admin->id && ! $aktif) {
            throw ValidationException::withMessages([
                'is_active' => 'Tidak bisa menonaktifkan akun sendiri.',
            ]);
        }

        $ownerHilang = $admin->role === 'owner'
            && (bool) $admin->is_active
            && ($role !== 'owner' || ! $aktif);

        if ($ownerHilang && ! $this->adaOwnerAktifLain($admin->id)) {
            throw ValidationException::withMessages([
                'role' => 'Minimal satu owner aktif harus ada.',
            ]);
        }

        $attributes = [
            'name' => trim((string) $data['name']),
            'email' => strtolower(trim((string) $data['email'])),
            'role' => $role,
            'is_active' => $aktif,
        ];

        if (isset($data['password']) && (string) $data['password'] !== '') {
            $attributes['password'] = (string) $data['password'];
        }

        $admin->forceFill($attributes)->save();

        return redirect()
            ->route('admin.accounts.index')
            ->with('success', 'Akun admin diperbarui.');
    }

    public function destroy(Request $request, Admin $admin): RedirectResponse
    {
        $me = $request->user('admin');

        if ($me !== null && $me->id === $admin->id) {
            throw ValidationException::withMessages([
                'akun' => 'Tidak bisa menghapus akun sendiri.',
            ]);
        }

        if ($admin->role === 'owner'
            && (bool) $admin->is_active
            && ! $this->adaOwnerAktifLain($admin->id)) {
            throw ValidationException::withMessages([
                'akun' => 'Minimal satu owner aktif harus ada.',
            ]);
        }

        $admin->delete();

        return redirect()
            ->route('admin.accounts.index')
            ->with('success', 'Akun admin dihapus.');
    }

    /**
     * @return array<string, mixed>
     */
    private function row(Admin $admin, ?int $viewerId): array
    {
        return [
            'id' => $admin->id,
            'name' => $admin->name,
            'email' => $admin->email,
            'role' => $admin->role,
            'is_active' => (bool) $admin->is_active,
            'is_self' => $viewerId !== null && $viewerId === $admin->id,
            'created_at' => $admin->created_at?->toIso8601String(),
        ];
    }

    private function adaOwnerAktifLain(int $adminId): bool
    {
        return Admin::query()
            ->where('role', 'owner')
            ->where('is_active', true)
            ->whereKeyNot($adminId)
            ->exists();
    }
}
