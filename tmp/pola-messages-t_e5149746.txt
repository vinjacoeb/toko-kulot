    18	/**

    24	 *   filter    { q: string|null, status: 'semua'|'belum'|'sudah' }
    25	 *   ringkasan { total, belum_dibaca }

    27	 *
    28	 * Aksi memakai URL literal routes/admin.php:
    29	 *   tandai dibaca -> PATCH /admin/pesan-masuk/{id}/baca  (kontrak §10.2, koreksi PATCH 20 Okt 2026)

    33	const BREADCRUMB = [{ label: 'Dashboard', href: '/admin' }, { label: 'Pesan Masuk' }];
    34	
    35	const DEFAULT_FILTER = { q: '', status: 'semua' };
    36	
    37	/** Enum `status` mengikuti kontrak §10.1: semua | belum | sudah. */
    38	const STATUS_OPTIONS = [
    39	    { value: 'semua', label: 'Semua status' },
    40	    { value: 'belum', label: 'Belum dibaca' },
    41	    { value: 'sudah', label: 'Sudah dibaca' },
    42	];
    43	
    44	/**
    45	 * Sidebar admin: rute /admin/pesan-masuk sudah terdaftar (routes/admin.php), jadi
    46	 * entri "Pesan" dibuka di halaman ini — pola sama dengan Settings/Index.jsx.
    47	 */
    48	const NAV_AKTIF = ADMIN_NAV.map((item) => (item.key === 'pesan' ? { ...item, disabled: false } : item));
    49	
    50	/** Nilai server bisa null; semua teks melewati helper ini agar tidak ada "undefined" di layar. */
    51	const asText = (value) => {
    52	    if (value === null || value === undefined) return '';
    53	
    54	    return typeof value === 'string' ? value : String(value);
    55	};
    56	
    57	/** Angka dari server bisa string; hasil selalu >= 0 dan bukan NaN. */
    58	const asCount = (value) => {
    59	    const number = Number(value ?? 0);
    60	
    61	    return Number.isFinite(number) && number > 0 ? number : 0;
    62	};
    63	
    64	/** Normalisasi props `filter` dari server; nilai kosong/tak dikenal -> bawaan. */
    65	const normalizeFilter = (filter) => ({
    66	    q: typeof filter?.q === 'string' ? filter.q : '',
    67	    status: filter?.status === null || filter?.status === undefined ? 'semua' : String(filter.status),
    68	});
    69	
    70	/**
    71	 * Paginator Laravel dikirim datar (`current_page`, `last_page`, `total`, `data`);
    72	 * kontrak §10.1 menuliskannya sebagai `links` + `meta`. Keduanya dibaca supaya
    73	 * halaman tidak bergantung pada satu bentuk saja.
    74	 */
    75	const readPaginator = (paginator) => {
    76	    const source = paginator ?? {};
    77	    const meta = source.meta && typeof source.meta === 'object' ? source.meta : source;
    78	
    79	    return {
    80	        rows: Array.isArray(source.data) ? source.data : [],
    81	        currentPage: Math.max(1, Number(meta.current_page) || 1),
    82	        lastPage: Math.max(1, Number(meta.last_page) || 1),
    83	        total: asCount(meta.total),
    84	    };
    85	};

    90	    filter = null,

    96	    const [navError, setNavError] = useState(null);
    97	    const [toast, setToast] = useState(null);
    98	    const [busyId, setBusyId] = useState(null);
    99	    const [actionError, setActionError] = useState(null);
   100	    const [targetId, setTargetId] = useState(null);
   101	    const [askDiscard, setAskDiscard] = useState(false);
   102

   298	        },
   299	        {
