import React from 'react';
import { cn } from '../lib/format';

/**
 * Pembungkus isian - docs/design.md 4.5.
 * Label wajib htmlFor; hint dan galat terhubung lewat aria-describedby.
 */
export default function Field({
    label,
    htmlFor,
    required = false,
    hint = null,
    error = null,
    children,
    className = '',
}) {
    const hintId = htmlFor ? htmlFor + '-hint' : undefined;
    const errorId = htmlFor ? htmlFor + '-error' : undefined;

    return (
        <div className={cn('flex flex-col', className)}>
            {label && (
                <label htmlFor={htmlFor} className="mb-1.5 block text-meta font-semibold text-ink">
                    {label}
                    {required && (
                        <span className="text-danger" aria-hidden="true">
                            {' *'}
                        </span>
                    )}
                </label>
            )}
            {children}
            {hint && !error && (
                <p id={hintId} className="mt-1.5 text-meta text-slate-400">
                    {hint}
                </p>
            )}
            {error && (
                <p id={errorId} className="mt-1.5 text-meta text-danger" role="alert">
                    {error}
                </p>
            )}
        </div>
    );
}

===== E. Button.jsx =====
import React from 'react';
import { Link } from '@inertiajs/react';
import { cn } from '../lib/format';
import { SpinnerIcon } from './icons';

const VARIANTS = {
    primary: 'btn-primary',
    secondary: 'btn-secondary',
    ghost: 'btn-ghost',
    danger:
        'inline-flex items-center justify-center gap-2 h-10 px-5 rounded-sm text-label font-semibold leading-none bg-danger text-white transition-colors duration-hover hover:opacity-90 disabled:opacity-45 disabled:cursor-not-allowed',
    link: 'inline-flex items-center gap-2 text-label text-brand underline underline-offset-4 rounded-sm',
};

const SIZES = {
    md: '',
    sm: 'h-10 px-4',
};

/**
 * Tombol baku (docs/komponen.md §4.1).
 * - `loading` menonaktifkan tombol dan menampilkan spinner (design §4.0).
 * - `href` mengubah tombol menjadi tautan Inertia.
 */
export default function Button({
    children,
    variant = 'primary',
    size = 'md',
    as = 'button',
    href = null,
    type = 'button',
    loading = false,
    disabled = false,
    fullWidth = false,
    icon = null,
    className = '',
    ...rest
}) {
    const classes = cn(
        VARIANTS[variant] ?? VARIANTS.primary,
        size === 'sm' && variant !== 'link' ? SIZES.sm : null,
        variant === 'link' && size === 'sm' ? 'text-meta' : null,
        fullWidth ? 'w-full' : null,
        'whitespace-nowrap',
        className
    );

    const content = loading ? (
        <>
            <SpinnerIcon size={16} />
            <span>Memproses…</span>
        </>
    ) : (
        <>
            {icon}
            <span>{children}</span>
        </>
    );

    if (as === 'a') {
        return (
            <a href={href} className={classes} {...rest}>
                {content}
            </a>
        );
    }

    if (as === Link || as === 'Link' || href) {
        return (
            <Link href={href} className={classes} aria-busy={loading || undefined} {...rest}>
                {content}
            </Link>
        );
    }

    return (
        <button
            type={type}
            className={classes}
            disabled={disabled || loading}
            aria-busy={loading || undefined}
            {...rest}
        >
            {content}
        </button>
    );
}

import React, { useEffect, useRef, useState } from 'react';
import { cn } from '../lib/format';
import { CloseIcon, SearchIcon } from './icons';

/** SearchInput - docs/design.md 4.19. Debounce 300ms sebelum memanggil onChange. */
export default function SearchInput({
    value = '',
    onChange = null,
    placeholder = 'Cari produk',
    label = 'Cari produk',
    loading = false,
    className = '',
}) {
    const [text, setText] = useState(value);
    const mounted = useRef(false);

    useEffect(() => {
        setText(value);
    }, [value]);

    useEffect(() => {
        if (!mounted.current) {
            mounted.current = true;
            return undefined;
        }
        const timer = window.setTimeout(() => {
            if (typeof onChange === 'function') onChange(text.trim());
        }, 300);
        return () => window.clearTimeout(timer);
    }, [text]);

    return (
        <div className={cn('relative', className)}>
            <span className="pointer-events-none absolute left-3.5 top-1/2 -translate-y-1/2 text-slate-400" aria-hidden="true">
                <SearchIcon width={18} height={18} />
            </span>
            <input
                type="search"
                value={text}
                onChange={(event) => setText(event.target.value)}
                placeholder={placeholder}
                aria-label={label}
                aria-busy={loading ? 'true' : undefined}
                className="h-field w-full rounded-md border border-line bg-white pl-10 pr-10 text-body text-ink transition-colors duration-hover placeholder:text-slate-400 focus:border-brand focus:outline-none focus:ring-[3px] focus:ring-brand/15"
            />
            {text.length > 0 && (
                <button
                    type="button"
                    onClick={() => setText('')}
                    aria-label="Bersihkan pencarian"
                    className="absolute right-2 top-1/2 -translate-y-1/2 rounded-sm p-1.5 text-slate-400 transition-colors duration-hover hover:text-ink focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-brand"
                >
                    <CloseIcon width={16} height={16} />
                </button>
            )}
        </div>
    );
}

function toArray(value) {
    if (Array.isArray(value)) return value.filter(Boolean);
    if (value === null || value === undefined || value === '') return [];
    return [value];
}

function Group({ title, children }) {
    return (
        <section className="border-b border-line py-4 first:pt-0 last:border-b-0">
            <h3 className="mb-2.5 text-meta font-semibold uppercase tracking-wide text-slate-400">{title}</h3>
            {children}
        </section>
    );
}

function FilterGroups({ draft, setDraft, categories, sizes }) {
    const ukuran = toArray(draft.ukuran);

    const toggleSize = (size) => {
        const next = ukuran.includes(size) ? ukuran.filter((item) => item !== size) : [...ukuran, size];
        setDraft({ ...draft, ukuran: next });
    };

    return (
        <div>
            <Group title="Kategori">
                <div className="flex flex-col gap-2">
                    <label className="flex items-center gap-2 text-body text-ink">
                        <input
                            type="radio"
                            name="kategori"
                            value=""
                            checked={!draft.kategori}
                            onChange={() => setDraft({ ...draft, kategori: '' })}
                            className="h-4 w-4 border-line text-brand focus:ring-brand"
                        />
                        Semua kategori
                    </label>
                    {(categories || []).map((category) => (
                        <label key={category.id ?? category.slug} className="flex items-center gap-2 text-body text-ink">
                            <input
                                type="radio"
                                name="kategori"
                                value={category.slug ?? ''}
                                checked={draft.kategori === (category.slug ?? '')}
                                onChange={() => setDraft({ ...draft, kategori: category.slug ?? '' })}
                                className="h-4 w-4 border-line text-brand focus:ring-brand"
                            />
                            <span className="flex-1">{category.name}</span>
                            {category.product_count !== undefined && (
                                <span className="text-meta text-slate-400">{formatNumber(category.product_count)}</span>
                            )}
                        </label>
                    ))}
                </div>
            </Group>

            <Group title="Harga">
                <div className="flex items-center gap-2">
                    <Field label="Minimum" htmlFor="harga-min" className="flex-1">
                        <Input
                            id="harga-min"
                            type="number"
                            inputMode="numeric"
                            min="0"
                            placeholder="0"
                            value={draft.harga_min ?? ''}
                            onChange={(event) => setDraft({ ...draft, harga_min: event.target.value })}
                        />
                    </Field>
                    <Field label="Maksimum" htmlFor="harga-max" className="flex-1">
                        <Input
                            id="harga-max"
                            type="number"
                            inputMode="numeric"
                            min="0"
                            placeholder="Tanpa batas"
                            value={draft.harga_max ?? ''}
                            onChange={(event) => setDraft({ ...draft, harga_max: event.target.value })}
                        />
                    </Field>
                </div>
                {(draft.harga_min || draft.harga_max) && (
                    <p className="mt-2 text-meta text-slate-400">
                        {rupiah(Number(draft.harga_min || 0)) + ' - ' + (draft.harga_max ? rupiah(Number(draft.harga_max)) : 'tanpa batas')}
                    </p>
                )}
            </Group>

            {sizes && sizes.length > 0 && (
                <Group title="Ukuran">
                    <div className="flex flex-wrap gap-2">
                        {sizes.map((size) => (
                            <VariantChip
                                key={size}
                                label={size}
                                active={ukuran.includes(size)}
                                onClick={() => toggleSize(size)}
                            />
                        ))}
                    </div>
                </Group>
            )}
        </div>
    );
}

/**
 * FilterPanel - docs/design.md 4.18.
 * Sidebar sticky di desktop (langsung menerapkan), bottom sheet di mobile (Terapkan).
 */
export default function FilterPanel({
    categories = [],
    sizes = [],
    value = {},
    onApply = null,
    onReset = null,
    activeCount = 0,
    className = '',
}) {
    const [draft, setDraft] = useState(value || {});
    const [open, setOpen] = useState(false);

    useEffect(() => {
        setDraft(value || {});
    }, [value]);

    const apply = (next) => {
        if (typeof onApply === 'function') onApply(next);
    };

    return (
        <>
            <aside className={cn('hidden lg:block', className)} aria-label="Filter produk">
                <div className="sticky top-[68px] rounded-md border border-line bg-white p-4">
                    <div className="mb-2 flex items-center justify-between">
                        <h2 className="text-h3 text-ink">Filter</h2>
                        {activeCount > 0 && typeof onReset === 'function' && (
                            <button
                                type="button"
                                onClick={onReset}
                                className="text-meta text-brand underline underline-offset-4"
                            >
                                Reset
                            </button>
                        )}
                    </div>
                    <FilterGroups
                        draft={draft}
                        setDraft={(next) => {
                            setDraft(next);
                            apply(next);
                        }}
                        categories={categories}
                        sizes={sizes}
                    />
                </div>
            </aside>

            <button
                type="button"
                onClick={() => setOpen(true)}
                className="inline-flex h-field items-center gap-2 rounded-md border border-line bg-white px-4 text-meta font-semibold text-ink lg:hidden"
            >
                Filter
                {activeCount > 0 && (
                    <span className="inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-brand-700 px-1.5 text-badge text-white">
                        {activeCount}
                    </span>
                )}
            </button>

            <BottomSheet
                open={open}
                onClose={() => setOpen(false)}
                title="Filter"
                footer={
                    <div className="flex gap-2">
                        <button
                            type="button"
                            onClick={() => {
                                const reset = {};
                                setDraft(reset);
                                if (typeof onReset === 'function') onReset();
                                setOpen(false);
                            }}
                            className="h-field flex-1 rounded-md border border-line text-meta font-semibold text-ink"
                        >
                            Reset
                        </button>
                        <button
                            type="button"
                            onClick={() => {
                                apply(draft);
                                setOpen(false);
                            }}
                            className="h-field flex-1 rounded-md bg-brand-700 text-meta font-semibold text-white"
                        >
                            Terapkan
                        </button>
                    </div>
                }
            >
                <FilterGroups draft={draft} setDraft={setDraft} categories={categories} sizes={sizes} />
            </BottomSheet>
        </>
    );
}

===== H. StatusBadge.jsx =====
import React from 'react';
import { cn } from '../lib/format';

/**
 * Pil status — docs/design.md §4.21.
 * Pemetaan baku status order (jangan bikin varian baru):
 * pending → warning, dibayar/diproses/dikirim → brand, selesai → success, batal → danger;
 * status di luar daftar (mis. "Sudah dibaca" §5.12, arsip) → neutral.
 * Bentuk kelas §4.21 v2.1: `bg-{token}/10 text-{token}-strong` — teks badge selalu < 24px
 * (`text-badge`) sehingga ambangnya 4,5:1, dan token dasar `warning`/`success` hanya 2,86:1/2,96:1
 * di atas tint-nya. `brand` tetap `bg-brand/10 text-brand` (9,35:1); `neutral` `bg-slate-200/60
 * text-slate-600` (6,17:1).
 */
const TONES = {
    pending: { token: 'warning', label: 'Menunggu' },
    dibayar: { token: 'brand', label: 'Dibayar' },
    diproses: { token: 'brand', label: 'Diproses' },
    dikirim: { token: 'brand', label: 'Dikirim' },
    selesai: { token: 'success', label: 'Selesai' },
    batal: { token: 'danger', label: 'Batal' },
};

const CLASSES = {
    brand: 'bg-brand/10 text-brand',
    warning: 'bg-warning/10 text-warning-strong',
    success: 'bg-success/10 text-success-strong',
    danger: 'bg-danger/10 text-danger-strong',
    neutral: 'bg-slate-200/60 text-slate-600',
};

export default function StatusBadge({ status = null, label = null, className = '' }) {
    const key = typeof status === 'string' ? status.trim().toLowerCase() : '';
    const tone = TONES[key] ?? null;
    const text = label ?? tone?.label ?? (key === '' ? '-' : status);

    return (
        <span
            className={cn(
                'inline-flex items-center rounded-full px-2.5 py-1 text-badge uppercase',
                CLASSES[tone?.token ?? 'neutral'],
                className
            )}
        >
            {text}
        </span>
    );
}
