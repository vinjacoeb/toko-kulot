## 4. Inventaris Komponen

Setiap komponen: struktur, ukuran, state. Wajib jadi komponen React reusable di `resources/js/Components/`. API props & checklist implementasi ada di `docs/komponen.md`.

### 4.0 Matriks state universal

Berlaku untuk **setiap** komponen yang mengambil data atau menerima aksi pengguna. QA menolak komponen yang tidak punya state di bawah.

| State | Pemicu | Tampilan wajib | Kelas/token acuan |
|---|---|---|---|
| `default` | data ada | konten normal | — |
| `loading` | request berjalan | skeleton mengikuti dimensi asli (bukan spinner penuh halaman), `aria-busy="true"` | `animate-pulse-slow` |
| `empty` | data kosong & bukan error | ilustrasi garis + teks penjelas + CTA keluar dari kondisi kosong | `text-slate-400`, §4.23 `EmptyState` |
| `error` | request gagal | pesan di dalam layout (bukan halaman putih) + tombol "Coba lagi" | `border-danger`, `text-danger`, §4.23 `ErrorState` |
| `disabled` | aksi tidak tersedia (stok 0, belum login) | opacity .45 + `cursor-not-allowed` + alasan terlihat | `disabled:opacity-45` |
| `focus-visible` | navigasi keyboard | ring 2px `brand` offset 2px | `ring-2 ring-brand ring-offset-2` |
| `hover` | pointer di atas elemen | sesuai komponen | `duration-hover` |

Aturan tambahan: `loading` yang melewati **10 detik** wajib berubah jadi `error` + tombol "Coba lagi" (mencegah layar menggantung); skeleton tidak boleh dipakai untuk menutupi error.

function Thumb({ src, size = 48, className = '' }) {
    const [failed, setFailed] = useState(false);

    useEffect(() => {
        setFailed(false);
    }, [src]);

    const style = { width: `${size}px`, height: `${size}px` };

    if (!src || failed) {
        return (
            <span
                style={style}
                className={cn('flex shrink-0 items-center justify-center rounded-md border border-line bg-surface text-slate-400', className)}
            >
                <ImageOffIcon size={18} />
            </span>
        );
    }

    return (
        <img
            src={src}
            alt=""
            style={style}
            loading="lazy"
            onError={() => setFailed(true)}
            className={cn('shrink-0 rounded-md border border-line object-cover', className)}
        />
    );
}

function ImageField({
    id,
    label,
    hint = null,
    required = false,
    accept,
    maxKb,
    extensions,
    file = null,
    currentPath = null,
    error = null,
    message,
    onSelect,
}) {
    const [objectUrl, setObjectUrl] = useState(null);

    useEffect(() => {
        if (!file) {
            setObjectUrl(null);
            return undefined;
        }
        const url = URL.createObjectURL(file);
        setObjectUrl(url);
        return () => URL.revokeObjectURL(url);
    }, [file]);

    const describedBy = error ? `${id}-error` : hint ? `${id}-hint` : undefined;
    const sizeLabel = maxKb >= 1024 ? `${Math.round(maxKb / 1024)} MB` : `${maxKb} KB`;

    return (
        <Field label={label} htmlFor={id} required={required} hint={hint} error={error}>
            <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:gap-4">
                <Thumb src={objectUrl || currentPath || null} size={112} className="rounded-lg" />
                <div className="flex min-w-0 flex-1 flex-col gap-2">
                    <input
                        id={id}
                        type="file"
                        accept={accept}
                        aria-describedby={describedBy}
                        aria-invalid={error ? 'true' : undefined}
                        onChange={(event) => {
                            const picked = event.target.files && event.target.files[0] ? event.target.files[0] : null;
                            onSelect(picked);
                            event.target.value = '';
                        }}

import React, { useEffect, useRef } from 'react';
import { cn } from '../lib/format';

import { CloseIcon } from './icons';

const SIZES = {
    md: 'max-w-modal',
    lg: 'max-w-modalLg',
};

/**
 * Modal dasar — docs/design.md §4.24 (turunan: `ConfirmDialog`).
 * Overlay gelap, Esc menutup, klik overlay menutup, fokus terperangkap,
 * scroll body dikunci, `aria-modal="true"`.
 */
export default function Modal({
    open = false,
    title = null,
    description = null,
    children = null,
    footer = null,
    onClose,
    size = 'md',
    closeLabel = 'Tutup',
}) {
    const panelRef = useRef(null);

    useBodyScrollLock(open);
    useFocusTrap(panelRef, open);

    useEffect(() => {
        if (!open) return undefined;

        const onKeyDown = (event) => {
            if (event.key === 'Escape') onClose?.();
        };

        document.addEventListener('keydown', onKeyDown);

        return () => document.removeEventListener('keydown', onKeyDown);
    }, [open, onClose]);

    if (!open) return null;

    return (
        <div
            className="fixed inset-0 z-modal flex items-center justify-center bg-black/50 p-4"
            onMouseDown={(event) => {
                if (event.target === event.currentTarget) onClose?.();
            }}
        >
            <div
                ref={panelRef}
                role="dialog"
                aria-modal="true"
                aria-labelledby={title ? 'modal-title' : undefined}
                tabIndex={-1}
                className={cn(
                    'w-full rounded-xl bg-white p-6 shadow-raised animate-modal-in',
                    SIZES[size] ?? SIZES.md
                )}
            >
                <div className="flex items-start justify-between gap-4">
                    <div className="min-w-0">
                        {title && (
                            <h2 id="modal-title" className="text-h3 text-ink">
                                {title}
                            </h2>
                        )}
                        {description && <p className="mt-1 text-meta text-ink-soft">{description}</p>}
                    </div>
                    {onClose && (
                        <button
                            type="button"
                            onClick={onClose}
                            aria-label={closeLabel}
                            className="-mr-2 -mt-1 inline-flex h-touch w-touch items-center justify-center rounded-md text-ink-soft transition-colors duration-hover hover:bg-surface"
                        >
                            <CloseIcon size={20} />
                        </button>
                    )}
                </div>

                {children && <div className="mt-4 text-body text-ink">{children}</div>}
                {footer && <div className="mt-6 flex flex-wrap justify-end gap-3">{footer}</div>}
            </div>
        </div>
    );
}
